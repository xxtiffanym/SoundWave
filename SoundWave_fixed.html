if v_run >= 5 then raise exception 'Please wait for an admin to reply first'; end if; end if; insert into public."APPEAL_MESSAGE" ("AppealID","SenderID","SenderRole","Body") values (v_app."AppealID", v_uid, 'Artist', v_body); -- ===== List appeals (admin: all, artist: own) ===== create or replace function public.appeal_overview() returns table ("AppealID" uuid, "SongID" uuid, "Title" text, "ArtistID" uuid, "ArtistName" text, "Status" text, "LastSender" text, "LastMessage" text, "UpdatedAt" timestamptz, "RejectReason" text, "SongStatus" text) language sql security definer set search_path = public as $$ select a."AppealID", a."SongID", s."Title"::text, a."ArtistID", coalesce(u."FullName", u."Username")::text, a."Status", a."LastSender", (select m."Body" from public."APPEAL_MESSAGE" m where m."AppealID" = a."AppealID" order by m."CreatedAt" desc limit 1), a."UpdatedAt", s."RejectReason", s."ApprovalStatus" from public."SONG_APPEAL" a join public."SONG" s on s."SongID" = a."SongID" left join public."USER" u on u."UserID" = a."ArtistID" where public.is_admin() or a."ArtistID" = auth.uid() order by a."UpdatedAt" desc; $$; -- ===== Admin decision on an appeal ===== create or replace function public.admin_resolve_appeal(p_song uuid, p_action text, p_note text default null) returns void language plpgsql security definer set search_path = public as $$ declare v_app record;
